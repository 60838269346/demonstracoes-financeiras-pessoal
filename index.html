function criarLinhaHorizontal(titulo, categoria, tipoCalculo, isGrupo = false, classeExtra = '') { const tr = document.createElement('tr'); if(isGrupo) tr.className = 'row-group'; if(classeExtra) tr.className += ' ' + classeExtra; let html = `${titulo}`; let valorMesAnteriorReal = null; mesesAno.forEach((mes, index) => { let valOrc = 0; let valReal = 0; if(tipoCalculo === 'categoria') { const catLower = categoria.trim().toLowerCase(); orcamentos.filter(o => o.mes === mes && o.categoria.trim().toLowerCase() === catLower).forEach(o => valOrc += o.valor); transacoes.filter(t => t.mes === mes && t.categoria.trim().toLowerCase() === catLower).forEach(t => valReal += t.valor); } else if(tipoCalculo === 'total_receitas') { orcamentos.filter(o => o.mes === mes && contasReceita.some(c => c.trim().toLowerCase() === o.categoria.trim().toLowerCase())).forEach(o => valOrc += o.valor); transacoes.filter(t => t.mes === mes && contasReceita.some(c => c.trim().toLowerCase() === t.categoria.trim().toLowerCase())).forEach(t => valReal += t.valor); } else if(tipoCalculo === 'total_despesas') { orcamentos.filter(o => o.mes === mes && contasDespesa.some(c => c.trim().toLowerCase() === o.categoria.trim().toLowerCase())).forEach(o => valOrc += o.valor); transacoes.filter(t => t.mes === mes && contasDespesa.some(c => c.trim().toLowerCase() === t.categoria.trim().toLowerCase())).forEach(t => valReal += t.valor); } else if(tipoCalculo === 'sobra_bruta') { let recOrc = 0, despOrc = 0, recReal = 0, despReal = 0; orcamentos.filter(o => o.mes === mes && contasReceita.some(c => c.trim().toLowerCase() === o.categoria.trim().toLowerCase())).forEach(o => recOrc += o.valor); orcamentos.filter(o => o.mes === mes && contasDespesa.some(c => c.trim().toLowerCase() === o.categoria.trim().toLowerCase())).forEach(o => despOrc += o.valor); transacoes.filter(t => t.mes === mes && contasReceita.some(c => c.trim().toLowerCase() === t.categoria.trim().toLowerCase())).forEach(t => recReal += t.valor); transacoes.filter(t => t.mes === mes && contasDespesa.some(c => c.trim().toLowerCase() === t.categoria.trim().toLowerCase())).forEach(t => despReal += t.valor); valOrc = recOrc - despOrc; valReal = recReal - despReal; } else if(tipoCalculo === 'total_semanal') { orcamentos.filter(o => o.mes === mes && contasSemanais.some(c => c.trim().toLowerCase() === o.categoria.trim().toLowerCase())).forEach(o => valOrc += o.valor); transacoes.filter(t => t.mes === mes && contasSemanais.some(c => c.trim().toLowerCase() === t.categoria.trim().toLowerCase())).forEach(t => valReal += t.valor); } else if(tipoCalculo === 'sobra_liquida') { let recOrc = 0, despOrc = 0, semOrc = 0, recReal = 0, despReal = 0, semReal = 0; orcamentos.filter(o => o.mes === mes && contasReceita.some(c => c.trim().toLowerCase() === o.categoria.trim().toLowerCase())).forEach(o => recOrc += o.valor); orcamentos.filter(o => o.mes === mes && contasDespesa.some(c => c.trim().toLowerCase() === o.categoria.trim().toLowerCase())).forEach(o => despOrc += o.valor); orcamentos.filter(o => o.mes === mes && contasSemanais.some(c => c.trim().toLowerCase() === o.categoria.trim().toLowerCase())).forEach(o => semOrc += o.valor); transacoes.filter(t => t.mes === mes && contasReceita.some(c => c.trim().toLowerCase() === t.categoria.trim().toLowerCase())).forEach(t => recReal += t.valor); transacoes.filter(t => t.mes === mes && contasDespesa.some(c => c.trim().toLowerCase() === t.categoria.trim().toLowerCase())).forEach(t => despReal += t.valor); transacoes.filter(t => t.mes === mes && contasSemanais.some(c => c.trim().toLowerCase() === t.categoria.trim().toLowerCase())).forEach(t => semReal += t.valor); valOrc = (recOrc - despOrc) - semOrc; valReal = (recReal - despReal) - semReal; } let variacaoPerc = 0; let temVariacao = false; if (index > 0 && valorMesAnteriorReal !== null && valorMesAnteriorReal !== 0) { variacaoPerc = ((valReal - valorMesAnteriorReal) / Math.abs(valorMesAnteriorReal)) * 100; temVariacao = true; } valorMesAnteriorReal = valReal; let estiloOrc = tipoCalculo.includes('sobra') || tipoCalculo.includes('total') ? `style="font-weight: bold; color: ${valOrc >= 0 ? '#16a34a' : '#ef4444'};"` : ''; let estiloReal = tipoCalculo.includes('sobra') || tipoCalculo.includes('total') ? `style="font-weight: bold; color: ${valReal >= 0 ? '#16a34a' : '#ef4444'};"` : ''; let textoDelta = "-"; let corDelta = ""; if (temVariacao) { textoDelta = (variacaoPerc > 0 ? "+" : "") + variacaoPerc.toFixed(1) + "%"; corDelta = variacaoPerc >= 0 ? "color: #16a34a;" : "color: #ef4444;"; } html += `R$ ${valOrc.toFixed(2)}`; html += `R$ ${valReal.toFixed(2)}`; html += `${textoDelta}`; }); tr.innerHTML = html; tabelaAnaliticaAno.appendChild(tr); }
